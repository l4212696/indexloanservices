<script>
  const form = document.getElementById("loanForm");
  const statusBox = document.getElementById("status");
  const submitBtn = document.getElementById("submitBtn");
  const methodSelect = document.getElementById("paymentMethod");

  document.getElementById("year").textContent = new Date().getFullYear();

  const tomorrow = new Date();
  tomorrow.setDate(tomorrow.getDate() + 1);
  document.getElementById("repayDate").min = tomorrow.toISOString().slice(0, 10);

  // Show wallet username field or bank fields depending on the selected method.
  function togglePaymentFields() {
    const isBank = methodSelect.value === "bank";
    const isWallet = methodSelect.value !== "" && !isBank;
    document.getElementById("walletBlock").hidden = !isWallet;
    document.getElementById("bankBlock").hidden = !isBank;
    document.getElementById("accountBlock").hidden = !isBank;
    document.getElementById("routingBlock").hidden = !isBank;
  }
  methodSelect.addEventListener("change", togglePaymentFields);

  function clearErrors() {
    form.querySelectorAll(".field-error").forEach(el => (el.textContent = ""));
    form.querySelectorAll(".invalid").forEach(el => el.classList.remove("invalid"));
    statusBox.className = "status";
    statusBox.textContent = "";
  }

  function showErrors(errors) {
    Object.entries(errors).forEach(([key, msg]) => {
      const holder = form.querySelector(`[data-for="${key}"]`);
      if (holder) holder.textContent = msg;
      const input = form.querySelector(`[name="${key}"]`);
      if (input) input.classList.add("invalid");
    });
    const first = Object.keys(errors)[0];
    const target = form.querySelector(`[name="${first}"]`);
    if (target && !target.closest("[hidden]")) target.focus();
  }

  form.addEventListener("submit", async (e) => {
    e.preventDefault();
    clearErrors();

    const fd = new FormData(form);
    const method = fd.get("paymentMethod") || "";
    const isBank = method === "bank";

    const payload = {
      fullName: fd.get("fullName"),
      email: fd.get("email"),
      amount: fd.get("amount"),
      repayDate: fd.get("repayDate"),
      referredBy: fd.get("referredBy"),
      reason: fd.get("reason"),
      paymentMethod: method,
      paymentHandle: isBank ? "" : fd.get("paymentHandle"),
      bankName: isBank ? fd.get("bankName") : "",
      accountNumber: isBank ? fd.get("accountNumber") : "",
      routingNumber: isBank ? fd.get("routingNumber") : "",
    };

    submitBtn.disabled = true;
    submitBtn.textContent = "Submitting...";

    try {
      const res = await fetch("/api/apply", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });
      const result = await res.json();

      if (!res.ok) {
        if (result.errors) showErrors(result.errors);
        statusBox.className = "status err";
        statusBox.textContent = result.error || "Something went wrong. Please try again.";
        return;
      }

      form.reset();
      togglePaymentFields();
      statusBox.className = "status ok";
      statusBox.textContent = result.message;
      window.scrollTo({ top: 0, behavior: "smooth" });
    } catch {
      statusBox.className = "status err";
      statusBox.textContent = "Network error. Please check your connection and try again.";
    } finally {
      submitBtn.disabled = false;
      submitBtn.textContent = "Submit application";
    }
  });
</script>
